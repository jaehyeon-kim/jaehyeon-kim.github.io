{{- /* Overrides the theme render hook to give each heading anchor an accessible name. The theme emits `` with no content, because the link icon comes from CSS. Lighthouse fails "Links do not have a discernible name" on every heading, which costs 3 accessibility points on every post page, and a keyboard or screen reader user reaches a link that announces nothing. aria-label carries the heading text, and the visible output is unchanged. */ -}} {{- $anchor := printf "" .Anchor (.PlainText | htmlEscape) | safeHTML -}} {{- $numberifyHeadings := true }} {{- if isset .Page.Params "numberifyheadings" }} {{- $numberifyHeadings = .Page.Params.numberifyHeadings }} {{- end }} {{ printf "%s%s" .Level .Anchor (cond $numberifyHeadings ` data-numberify` ``) .PlainText $anchor .Level | safeHTML }}
